cmake_minimum_required(VERSION 3.20)
project(xfce4-copilot-prompt-panel VERSION 0.1.0 LANGUAGES C)

include(GNUInstallDirs)
include(CTest)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PANEL REQUIRED IMPORTED_TARGET libxfce4panel-2.0>=4.18)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0>=3.24)
pkg_check_modules(GMODULE REQUIRED IMPORTED_TARGET gmodule-2.0)
pkg_check_modules(YAML REQUIRED IMPORTED_TARGET yaml-0.1)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Werror)

add_library(copilot-ui STATIC src/copilot-ui.c src/copilot-sessions.c)
set_target_properties(copilot-ui PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(copilot-ui PUBLIC src)
target_link_libraries(copilot-ui PUBLIC PkgConfig::GTK PkgConfig::PANEL PkgConfig::YAML)

add_library(copilot MODULE src/copilot-plugin.c)
target_link_libraries(copilot PRIVATE copilot-ui)
install(TARGETS copilot LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/xfce4/panel/plugins)
configure_file(data/copilot.desktop.in copilot.desktop @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/copilot.desktop
        DESTINATION ${CMAKE_INSTALL_DATADIR}/xfce4/panel/plugins)

add_executable(copilot-panel-preview src/preview.c)
target_link_libraries(copilot-panel-preview PRIVATE copilot-ui)

if(BUILD_TESTING)
  pkg_check_modules(JSON REQUIRED IMPORTED_TARGET json-glib-1.0)
  add_executable(fake-copilot tests/fake-copilot.c)
  target_link_libraries(fake-copilot PRIVATE PkgConfig::JSON)
  add_executable(test-ui tests/test-ui.c)
  target_link_libraries(test-ui PRIVATE copilot-ui PkgConfig::JSON)
  add_test(NAME ui COMMAND test-ui $<TARGET_FILE:fake-copilot>)
  set_tests_properties(ui PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77)
  add_executable(test-panel tests/test-panel.c)
  target_link_libraries(test-panel PRIVATE PkgConfig::GTK PkgConfig::PANEL PkgConfig::GMODULE)
  add_test(NAME panel COMMAND test-panel $<TARGET_FILE:copilot>)
  set_tests_properties(panel PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77)
  add_executable(test-sessions tests/test-sessions.c)
  target_link_libraries(test-sessions PRIVATE copilot-ui)
  add_test(NAME sessions COMMAND test-sessions)
endif()
